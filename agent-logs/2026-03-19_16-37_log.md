## 1. Cone Analytic Jacobian (Gauss-Newton in `fit_cone()`)

From [surface_fit.cpp](../brepper-cpp2/src/surface_fit.cpp#L186-L227):

```cpp
// Gauss-Newton refinement of all cone parameters to minimise vertex-to-cone
// distances.  Only run with sufficient data to avoid overfitting.
// Parameters: α (half_angle), t (h_apex), q_u/q_w (axis lateral position),
//             δ_u/δ_w (axis direction perturbations).
//   f_i = r_i * cos(α) - |h_raw_i - t| * sin(α)
if (nv >= 10) {
    Eigen::Vector3d gn_u, gn_w;
    Eigen::VectorXd lat_u(nv), lat_w(nv);
    for (int iter = 0; iter < 20; iter++) {
        // Recompute orthonormal basis and vertex geometry for current params
        const Eigen::Vector3d arb2 = (std::abs(axis_dir.x()) < 0.9)
            ? Eigen::Vector3d(1, 0, 0) : Eigen::Vector3d(0, 1, 0);
        gn_u = axis_dir.cross(arb2).normalized();
        gn_w = axis_dir.cross(gn_u).normalized();
        for (int i = 0; i < nv; i++) {
            const Eigen::Vector3d diff = P.row(i).transpose() - Q_perp;
            h_raw(i) = diff.dot(axis_dir);
            const Eigen::Vector3d lat = diff - h_raw(i) * axis_dir;
            r_raw(i) = lat.norm();
            lat_u(i) = lat.dot(gn_u);
            lat_w(i) = lat.dot(gn_w);
        }
        const double ca = std::cos(half_angle);
        const double sa = std::sin(half_angle);
        Eigen::MatrixXd J(nv, 6);
        Eigen::VectorXd f(nv);
        for (int i = 0; i < nv; i++) {
            const double hi = h_raw(i) - h_apex;
            const double abs_hi = std::abs(hi);
            const double sign_hi = (hi >= 0) ? 1.0 : -1.0;
            const double ri = std::max(r_raw(i), 1e-12);
            f(i) = r_raw(i) * ca - abs_hi * sa;
            J(i, 0) = -r_raw(i) * sa - abs_hi * ca;              // ∂f/∂α
            J(i, 1) = sign_hi * sa;                               // ∂f/∂t
            J(i, 2) = -ca * lat_u(i) / ri;                       // ∂f/∂q_u
            J(i, 3) = -ca * lat_w(i) / ri;                       // ∂f/∂q_w
            const double axis_sens = ca * h_raw(i) / ri + sign_hi * sa;
            J(i, 4) = -lat_u(i) * axis_sens;                     // ∂f/∂δ_u
            J(i, 5) = -lat_w(i) * axis_sens;                     // ∂f/∂δ_w
        }
        const Eigen::VectorXd delta = J.colPivHouseholderQr().solve(-f);
        half_angle += delta(0);
        h_apex += delta(1);
        Q_perp += delta(2) * gn_u + delta(3) * gn_w;
        axis_dir = (axis_dir + delta(4) * gn_u + delta(5) * gn_w).normalized();
        if (delta.norm() < 1e-14) break;
    }
}
```

```cpp
        const Eigen::Vector3d gn_u = axis_dir.cross(arb2).normalized();
        const Eigen::Vector3d gn_w = axis_dir.cross(gn_u).normalized();

        Eigen::MatrixXd J(nv, 7);
        Eigen::VectorXd f(nv);
        for (int i = 0; i < nv; i++) {
            const Eigen::Vector3d diff = P.row(i).transpose() - center;
            const double h_i = diff.dot(axis_dir);
            const Eigen::Vector3d lat = diff - h_i * axis_dir;
            const double rho_i = lat.norm();  // radial distance from axis
            const double dr = rho_i - R;
            const double tube_dist = std::sqrt(dr * dr + h_i * h_i);
            if (tube_dist < 1e-12) {
                J.row(i).setZero();
                f(i) = -r;
                continue;
            }
            f(i) = tube_dist - r;

            const double s_r = dr / tube_dist;
            const double s_h = h_i / tube_dist;

            Eigen::Vector3d rho_hat = Eigen::Vector3d::Zero();
            if (rho_i > 1e-12) rho_hat = lat / rho_i;

            // ∂f/∂center = -(s_r * rho_hat + s_h * axis_dir)
            const Eigen::Vector3d df_ddiff = s_r * rho_hat + s_h * axis_dir;
            J(i, 0) = -df_ddiff.x();
            J(i, 1) = -df_ddiff.y();
            J(i, 2) = -df_ddiff.z();

            // ∂f/∂R = -s_r
            J(i, 3) = -s_r;

            // ∂f/∂r = -1
            J(i, 4) = -1.0;

            // ∂f/∂δ_u, ∂f/∂δ_w: axis perturbations
            const double dh_du = diff.dot(gn_u);
            const double dh_dw = diff.dot(gn_w);
            double drho_du = 0, drho_dw = 0;
            if (rho_i > 1e-12) {
                drho_du = -h_i * gn_u.dot(rho_hat);
                drho_dw = -h_i * gn_w.dot(rho_hat);
            }
            J(i, 5) = s_r * drho_du + s_h * dh_du;
            J(i, 6) = s_r * drho_dw + s_h * dh_dw;
        }
        const Eigen::VectorXd delta = J.colPivHouseholderQr().solve(-f);
        center += Eigen::Vector3d(delta(0), delta(1), delta(2));
        R += delta(3);
        r += delta(4);
        axis_dir = (axis_dir + delta(5) * gn_u + delta(6) * gn_w).normalized();
        if (delta.norm() < 1e-14) break;
    }
}
```

**Residual**: $f_i = \sqrt{(\rho_i - R)^2 + h_i^2} - r$ where $\rho_i$ is radial distance from axis and $h_i$ is axial distance from center.

## 3. Torus Minor-Circle Acceptance Test

From [stage2.cpp](../brepper-cpp2/src/stage2.cpp#L808-L852):

```cpp
static bool acceptance_test(const ConnectedMesh& mesh,
                            const std::vector<int>& face_list,
                            const Fit& fit,
                            const Stage2Config& /*config*/)
{
    // Minimum 7 faces
    if (face_list.size() < 7) return false;

    // Angular coverage around the minor circle (tube cross-section).
    // For each face centroid, compute the angle in the (radial, axial) plane
    // relative to the tube center at that azimuth.
    std::vector<double> minor_angles;
    minor_angles.reserve(face_list.size());
    for (const int fi : face_list) {
        const Eigen::Vector3d c = face_centroid(mesh, fi);
        const Eigen::Vector3d diff = c - fit.center;
        const double axial = diff.dot(fit.axis_direction);
        const Eigen::Vector3d radial_vec = diff - axial * fit.axis_direction;
        const double radial_dist = radial_vec.norm();
        if (radial_dist < 1e-10) continue;
        // Direction from tube center to surface point in the cross-section plane
        const double dr = radial_dist - fit.major_radius;  // radial offset from tube center
        const double phi = std::atan2(axial, dr);
        minor_angles.push_back(phi);
    }

    if (minor_angles.size() < 7) return false;

    std::sort(minor_angles.begin(), minor_angles.end());
    const int n = static_cast<int>(minor_angles.size());

    // Compute largest gap
    double largest_gap = minor_angles[0] + 2.0 * M_PI - minor_angles[n - 1];
    for (int i = 1; i < n; i++) {
        const double gap = minor_angles[i] - minor_angles[i - 1];
        if (gap > largest_gap) largest_gap = gap;
    }

    const double span = 2.0 * M_PI - largest_gap;
    // Require meaningful angular coverage around the minor circle
    if (span < 0.1) return false;

    return true;
}
```

Computes $\phi = \text{atan2}(\text{axial}, \rho - R)$ for each face centroid, then checks that the angular span around the tube cross-section is at least 0.1 radians (~5.7°). Note: unlike the conical acceptance test, this does **not** check second-largest gap.

## 4. Vertex-Position PCA for Torus Axis Estimation

From [surface_fit.cpp](../brepper-cpp2/src/surface_fit.cpp#L362-L381):

```cpp
// Step 1: Axis direction from PCA of vertex positions.
// For a torus, the direction of minimum positional variance is the axis.
// This works because axis variance ~ r^2/2 while perpendicular variance ~
// (R^2 + r^2/2)/2 >> r^2/2 for any non-self-intersecting torus (R > r).
// (Unlike cylinders/cones where face normals are perpendicular to the axis,
// torus normals have significant axis components, so N^T N doesn't give
// the axis as its smallest eigenvector.)
const Eigen::Vector3d p_mean = P.colwise().mean();
const Eigen::MatrixXd Pc = P.rowwise() - p_mean.transpose();
const Eigen::Matrix3d PtP = Pc.transpose() * Pc;
Eigen::SelfAdjointEigenSolver<Eigen::Matrix3d> eig(PtP);
Eigen::Vector3d axis_dir = eig.eigenvectors().col(0);
axis_dir.normalize();
```

The key insight (commented in the code): for a torus with $R > r$, the axis direction has the **smallest** positional variance ($\sim r^2/2$), while perpendicular directions have variance $\sim (R^2 + r^2/2)/2$. This is why PCA of vertex positions works, as opposed to PCA of face normals which works for cylinders/cones but not tori.
