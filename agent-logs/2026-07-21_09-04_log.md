230	/// Resolve multi-claimed faces to their best-fitting claimant (2× margin to
231	/// leave the larger claimant), rebuild each hypothesis's face/vertex lists,
232	/// drop hypotheses that lost all faces, and collect unclaimed leftovers.
233	fn resolve_claims(
234	    mesh: &ConnectedMesh,
235	    region_faces: &[usize],
236	    committed: &mut Vec<CompeteHyp>,
237	    claim_lists: &[Vec<usize>],
238	) -> (Vec<CompeteHyp>, Vec<usize>) {
239	    let hyp_area: Vec<f64> = committed
240	        .iter()
241	        .map(|h| h.faces().iter().map(|&fi| face_area(&mesh.faces[fi], &mesh.vertices)).sum())
242	        .collect();
243	
244	    let mut assigned: Vec<Vec<usize>> = vec![Vec::new(); committed.len()];
245	    let mut leftovers: Vec<usize> = Vec::new();
246	
247	    for &fi in region_faces {
248	        let claimants = &claim_lists[fi];
249	        if claimants.is_empty() {
250	            leftovers.push(fi);
251	            continue;
252	        }
253	        if claimants.len() == 1 {
254	            assigned[claimants[0]].push(fi);
255	            continue;
256	        }
257	        // Best (smallest max-vertex-distance) claimant; the larger-area
258	        // claimant wins ties and only loses by a 2× margin (a true junction
259	        // face fits both sides equally — the margin prevents flapping).
260	        let dist = |ci: usize| max_vertex_dist(mesh, &committed[ci], fi);
261	        let mut order: Vec<usize> = claimants.clone();
262	        order.sort_by(|&a, &b| {
263	            hyp_area[b]
264	                .partial_cmp(&hyp_area[a])
265	                .unwrap()
266	                .then_with(|| a.cmp(&b))
267	        });
268	        let incumbent = order[0]; // largest area
269	        let mut winner = incumbent;
270	        let mut winner_d = dist(incumbent);
271	        for &ci in &order[1..] {
272	            let d = dist(ci);
273	            if d * 2.0 < winner_d {
274	                winner = ci;
275	                winner_d = d;
276	            }
277	        }
278	        assigned[winner].push(fi);
279	    }
280	
281	    let mut out = Vec::new();
282	    for (ci, mut hyp) in committed.drain(..).enumerate() {
283	        let faces = std::mem::take(&mut assigned[ci]);
284	        if faces.is_empty() {
285	            continue;
286	        }
287	        let verts = vertices_of(mesh, &faces);
288	        hyp.set_faces(faces, verts);
289	        out.push(hyp);
290	    }
291	    (out, leftovers)
292	}

294	/// Grow a hypothesis of type \`S\` from \`seed\`: condition an initial fit on a
295	/// jet-similar seed patch, then flood-accept every region face the fit
296	/// validates and refit on the grown support, to a fixpoint. Returns the
297	/// hypothesis, its validated area, and its face list.
298	fn grow<S: SurfaceType>(
299	    mesh: &ConnectedMesh,
300	    in_region: &[bool],
301	    seed: usize,
302	    jet: &[JetField],
303	    centroids: &[[f64; 3]],
304	    config: &Config,
305	    min_faces: usize,
306	) -> Option<(S::Hypothesis, f64, Vec<usize>)> {

308	    let stol = config.surface_tolerance_mm;

311	    let mut prev = 0usize;

315	        S::determine_convexity(mesh, &accepted, &mut fit);
316	        let comp = flood::<S>(mesh, in_region, seed, &fit, centroids, vtol, stol);
317	        if comp.is_empty() {

321	            accepted = comp;

326	            if comp.len() >= accepted.len() {

349	/// Connected region-faces reachable from \`seed\` that all validate \`fit\`
350	/// (vertex tolerance + centroid tolerance + orientation coherence).
351	fn flood<S: SurfaceType>(
352	    mesh: &ConnectedMesh,
353	    in_region: &[bool],
354	    seed: usize,
355	    fit: &S::Fit,
356	    centroids: &[[f64; 3]],
357	    vtol: f64,
358	    stol: f64,
359	) -> Vec<usize> {
360	    if !face_validates::<S>(mesh, seed, fit, centroids, vtol, stol) {
361	        return Vec::new();
362	    }
363	    let mut seen: HashSet<usize> = HashSet::new();
364	    seen.insert(seed);
365	    let mut out = Vec::new();
366	    let mut stack = vec![seed];
367	    while let Some(fi) = stack.pop() {
368	        out.push(fi);
369	        let vc = mesh.faces[fi].vertex_count as usize;
370	        for &ni in &mesh.faces[fi].neighbors[..vc] {
371	            if ni < 0 {

378	            {
379	                seen.insert(ni);
380	                stack.push(ni);
381	            }
382	        }
383	    }
384	    out.sort_unstable();
385	    out
386	}

388	/// Does face \`fi\` validate \`fit\` (all vertices ≤ vtol, centroid ≤ stol,
389	/// orientation coherent)?
390	fn face_validates<S: SurfaceType>(
391	    mesh: &ConnectedMesh,
392	    fi: usize,
393	    fit: &S::Fit,
394	    centroids: &[[f64; 3]],
395	    vtol: f64,
396	    stol: f64,
397	) -> bool {

399	        return false;
400	    }

412	/// A small jet-similar patch around \`seed\` (same curvature class, magnitude
413	/// within 2×), capped, for a well-conditioned initial fit.
414	fn seed_patch(mesh: &ConnectedMesh, in_region: &[bool], seed: usize, jet: &[JetField]) -> Vec<usize> {
415	    let seed_class = curvature_class(jet[seed]);
416	    let seed_mag = jet[seed].k1.abs().max(jet[seed].k2.abs());
417	    let mut seen: HashSet<usize> = HashSet::new();
418	    seen.insert(seed);
419	    let mut out = vec![seed];
420	    // Depth-limited BFS (frontier by ring).
421	    let mut frontier = vec![seed];

424	            break;
425	        }
426	        let mut next = Vec::new();

428	            let vc = mesh.faces[fi].vertex_count as usize;

432	                }

434	                if !in_region[ni] || seen.contains(&ni) {
435	                    continue;
436	                }
437	                if curvature_class(jet[ni]) != seed_class {
438	                    continue;
439	                }
440	                let mag = jet[ni].k1.abs().max(jet[ni].k2.abs());

/// Grow a toroidal candidate from an initial fit patch, trying flood starts
/// in order until one grows. Rejects degenerate giant-major-radius fits: a
/// short segment of straight cylinder fits a torus with an absurdly distant
/// axis (observed R ≈ 1500–3600 mm on ~100 mm models under the old BFS), and
/// a legitimate torus's major radius cannot exceed the mesh extent — the
/// same guard as the ladder's acceptance test.
fn torus_candidate(
    mesh: &ConnectedMesh,
    in_region: &[bool],
    initial: &[usize],
    starts: &[usize],
    centroids: &[[f64; 3]],
    config: &Config,
) -> Option<(ToroidalHypothesis, f64, Vec<usize>)> {
    let bb_diag = bounding_box_diagonal(&mesh.vertices);
    for &start in starts.iter().take(MAX_TORUS_STARTS) {
        if let Some((hyp, area, faces)) = grow_from::<ToroidalSurface>(
            mesh,
            in_region,
            start,
            initial.to_vec(),
            centroids,
            config,
            MIN_TORUS_FACES,
        ) {
            if hyp.major_radius <= bb_diag {
                return Some((hyp, area, faces));
            }
        }
    }
    None
}

/// Stage 2.3(c) bent-tube promotion. A cylinder flood that stalled against
/// region faces whose jet continues the tube — κ1 ≈ the tube's 1/r, |κ2|
/// well below κ1 (the same tube swinging around a bend, not a cap plane's
/// (0,0) or an end sphere's (κ,κ)) — is the developable prefix of a bent
/// tube. Fit a torus to the pure-bend patch just past the stall line
/// (frontier faces plus their off-cylinder tube-jet neighbours) and grow it;
/// the result competes on validated area like any other candidate.
///
/// The cylinder fit itself locates the tangency — frontier faces failed
/// cylinder validation — so unlike a jet-class seed patch (which cannot tell
/// bend from straight tube: both read the same κ1) this patch cannot
/// straddle the junction.
fn promote_bend(
    mesh: &ConnectedMesh,
    in_region: &[bool],
    cyl: &ConicalHypothesis,
    cyl_faces: &[usize],
    jet: &[JetField],
    centroids: &[[f64; 3]],
    config: &Config,
) -> Option<(ToroidalHypothesis, f64, Vec<usize>)> {
    if cyl.radius_at_apex <= 0.0 {
        return None;
    }
    let k_tube = 1.0 / cyl.radius_at_apex;
    let grow_vtol = config.vertex_tolerance_mm * GROW_VTOL_FACTOR;

    // Max vertex distance to the stalled cylinder — the "past the tangency"
    // test (frontier faces failed the flood at exactly this tolerance).
    let off_cyl_dist = |fi: usize| -> f64 {
        let face = &mesh.faces[fi];
        let vc = face.vertex_count as usize;
        face.vertex_indices[..vc]
            .iter()
            .map(|&vi| {
                super::cylindrical::vertex_to_cylinder_distance(
                    &mesh.vertices[vi],
                    &cyl.apex,
                    &cyl.axis_direction,
                    cyl.radius_at_apex,
                )
                .abs()
            })
            .fold(0.0_f64, f64::max)
    };
    // Tube-continuation jet signature. The κ2 bound is generous: a bend's
    // inner (saddle) side reads |κ2| = 1/(R−r), which approaches κ1 = 1/r
    // only for very tight bends; junction rows have contaminated jets and
    // are excluded from the fit patch either way (the flood claims them).
    let tube_jet = |fi: usize| -> bool {
        let a1 = jet[fi].k1.abs();
        a1 >= 0.5 * k_tube && a1 <= 2.0 * k_tube && jet[fi].k2.abs() <= 0.8 * a1
    };

    // Frontier: in-region neighbours of the cylinder support that are past
    // the tangency and continue the tube.
    let mut seen: HashSet<usize> = cyl_faces.iter().copied().collect();
    let mut frontier: Vec<usize> = Vec::new();
    for &fi in cyl_faces {
        let vc = mesh.faces[fi].vertex_count as usize;
        for &ni in &mesh.faces[fi].neighbors[..vc] {
            if ni < 0 {
                continue;
            }
            let ni = ni as usize;
            if !in_region[ni] || !seen.insert(ni) {
                continue;
            }
            if off_cyl_dist(ni) > grow_vtol && tube_jet(ni) {
                frontier.push(ni);
            }
        }
    }
    if frontier.is_empty() {
        return None; // finished a straight pipe (or stalled at a cap/sphere)
    }
    frontier.sort_unstable();

    // Pure-bend fit patch: depth-limited multi-source BFS from the frontier
    // through off-cylinder tube-jet faces. (Neighbours examined and rejected
    // above stay excluded — junction rows with contaminated jets would only
    // pollute the fit.)
    let mut patch: Vec<usize> = frontier.clone();
    let mut ring = frontier.clone();
    for _depth in 0..BEND_PATCH_DEPTH {
        if patch.len() >= BEND_PATCH_CAP {
            break;
        }
        let mut next: Vec<usize> = Vec::new();
        for &fi in &ring {
            let vc = mesh.faces[fi].vertex_count as usize;
            for &ni in &mesh.faces[fi].neighbors[..vc] {
                if ni < 0 {
                    continue;
                }
                let ni = ni as usize;
                if !in_region[ni] || seen.contains(&ni) {
                    continue;
                }
                if off_cyl_dist(ni) <= grow_vtol || !tube_jet(ni) {
                    continue;
                }
                seen.insert(ni);
                patch.push(ni);
                next.push(ni);
                if patch.len() >= BEND_PATCH_CAP {
                    break;
                }
            }
        }
        if next.is_empty() {
            break;
        }
        ring = next;
    }
    patch.sort_unstable();

    // Flood starts: deepest-off-cylinder first — the faces most surely
    // interior to the bend (frontier faces sit on the junction row and may
    // not validate the torus fit themselves).
    let mut starts = patch.clone();
    starts.sort_by(|&a, &b| {
        off_cyl_dist(b)
            .partial_cmp(&off_cyl_dist(a))
            .unwrap()
            .then_with(|| a.cmp(&b))
    });

    torus_candidate(mesh, in_region, &patch, &starts, centroids, config)
}
